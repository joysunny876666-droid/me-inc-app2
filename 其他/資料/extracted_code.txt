/**
 * Returns a formatted recurrence display string: "M/D | 每 N 週 | 下次: M/D"
 */
function getRecurrenceDisplayInfo(task, currentDateStr) {
    if (!task.recurrence) return '重複';
    const rec = task.recurrence;
    const startStr = rec.startDate || (task.createdAt ? task.createdAt.split('T')[0] : '1970-01-01');
    const startDate = new Date(startStr + 'T00:00:00');
    const sMmdd = `${startDate.getMonth() + 1}/${startDate.getDate()}`;

    // Build interval label
    const interval = rec.interval || 1;
    const typeMap = { daily: '日', weekly: '週', monthly: '月' };
    const typeLabel = typeMap[rec.type] || rec.type;
    const intervalLabel = `每 ${interval} ${typeLabel}`;

    // Calculate next occurrence after currentDateStr
    const nextStr = getNextOccurrenceStr(task, currentDateStr);
    const nextLabel = nextStr ? (() => {
        const nd = new Date(nextStr + 'T00:00:00');
        return `下次: ${nd.getMonth() + 1}/${nd.getDate()}`;
    })() : '';

    return [sMmdd, intervalLabel, nextLabel].filter(Boolean).join(' | ');
}

/**
 * Find the next occurrence date of a recurring task after `fromDateStr`.
 * Looks up to 366 days ahead.
 */
function getNextOccurrenceStr(task, fromDateStr) {
    if (!task.recurrence) return null;
    const rec = task.recurrence;
    const endStr = rec.endDate;
    const d = new Date(fromDateStr + 'T00:00:00');
    for (let i = 1; i <= 366; i++) {
        d.setDate(d.getDate() + 1);
        const testStr = `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}-${String(d.getDate()).padStart(2,'0')}`;
        if (endStr && testStr > endStr) break;
        // Reuse checkRecurrence logic inline
        const startStr = rec.startDate || '1970-01-01';
        if (testStr < startStr) continue;
        const startDate = new Date(startStr + 'T00:00:00');
        const targetDate = new Date(testStr + 'T00:00:00');
        const diffDays = Math.round((targetDate - startDate) / 86400000);
        const interval = rec.interval || 1;
        let matches = false;
        if (rec.type === 'daily') {
            matches = diffDays % interval === 0;
        } else if (rec.type === 'weekly') {
            if (rec.daysOfWeek && rec.daysOfWeek.length > 0) {
                if (rec.daysOfWeek.includes(targetDate.getDay())) {
                    matches = Math.floor(diffDays / 7) % interval === 0;
                }
            } else {
                matches = diffDays % (7 * interval) === 0;
            }
        } else if (rec.type === 'monthly') {
            if (targetDate.getDate() === startDate.getDate()) {
                const monthDiff = (targetDate.getFullYear() - startDate.getFullYear()) * 12 + (targetDate.getMonth() - startDate.getMonth());
                matches = monthDiff % interval === 0;
            }
        }
        if (matches) return testStr;
    }
    return null;
}

    timedTasks.forEach(task => {
        const hour = parseInt((task.time || '').split(':')[0]);

                block.addEventListener('click', () => openTimeRangeModal(task, todayStr, task.time));

function cleanupDuplicateTasks(oldName, newName, keepId) {
    if (!state.tasks) return;
    const keepTask = state.tasks.find(t => t.id === keepId);
    if (!keepTask) return;

    const namesToClean = new Set([oldName.trim().toLowerCase(), newName.trim().toLowerCase()]);

    state.tasks = state.tasks.filter(t => {
        if (t.id === keepId) return true;
        if (namesToClean.has(t.name.trim().toLowerCase())) {
            console.log(`Removing duplicate task "${t.name}" (ID: ${t.id}) to completely replace it.`);
            if (t.completedHistory) {
                if (!keepTask.completedHistory) keepTask.completedHistory = {};
                Object.keys(t.completedHistory).forEach(d => {
                    if (t.completedHistory[d]) {
                        keepTask.completedHistory[d] = true;
                    }
                });
            }
            return false;
        }
        return true;
    });
}

function cleanupAllDuplicateTasks() {
    if (!state.tasks) return false;
    
    const groups = {};
    state.tasks.forEach(t => {
        const key = t.name.trim().toLowerCase();
        if (!groups[key]) groups[key] = [];
        groups[key].push(t);
    });

    let changed = false;

    Object.keys(groups).forEach(key => {
        const list = groups[key];
        if (list.length <= 1) return;

        // Sort: Active recurring task first, then scheduled, then ones with end dates.
        list.sort((a, b) => {
            const aHasEnd = a.recurrence && a.recurrence.endDate;
            const bHasEnd = b.recurrence && b.recurrence.endDate;
            if (aHasEnd && !bHasEnd) return 1;
            if (!aHasEnd && bHasEnd) return -1;
            
            const aIsRec = a.type === 'recurring';
            const bIsRec = b.type === 'recurring';
            if (aIsRec && !bIsRec) return -1;
            if (!aIsRec && bIsRec) return 1;
            
            return b.id - a.id;
        });

        const keepTask = list[0];
        const duplicates = list.slice(1);

        console.log(`Auto-cleanup duplicates: Keeping task "${keepTask.name}" (ID: ${keepTask.id})`);

        if (!keepTask.completedHistory) keepTask.completedHistory = {};
        if (!keepTask.badHabitHistory) keepTask.badHabitHistory = {};

        duplicates.forEach(dup => {
            console.log(`Removing duplicate task "${dup.name}" (ID: ${dup.id})`);
            if (dup.completedHistory) {
                Object.keys(dup.completedHistory).forEach(d => {
                    if (dup.completedHistory[d]) {
                        keepTask.completedHistory[d] = true;
                    }
                });
            }
            if (dup.badHabitHistory) {
                Object.keys(dup.badHabitHistory).forEach(d => {
                    if (dup.badHabitHistory[d]) {
                        keepTask.badHabitHistory[d] = dup.badHabitHistory[d];
                    }
                });
            }
            state.tasks = state.tasks.filter(t => t.id !== dup.id);
            changed = true;
        });
    });

    return changed;
